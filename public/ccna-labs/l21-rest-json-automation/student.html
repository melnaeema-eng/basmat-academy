server.py

from http.server import BaseHTTPRequestHandler, HTTPServer
import json
STATE = {"g0/0": {"enabled": True, "description": "USERS"}}
TOKEN = "training-token"
class API(BaseHTTPRequestHandler):
    def log_message(self, *args): pass
    def reply(self, code, payload):
        body = json.dumps(payload).encode()
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)
    def request(self):
        if self.headers.get("Authorization") != "Bearer " + TOKEN:
            return self.reply(401, {"error": "unauthorized"})
        if self.path != "/api/interfaces":
            return self.reply(404, {"error": "not found"})
        if self.command == "GET":
            return self.reply(200, {"interfaces": STATE})
        try:
            n = int(self.headers.get("Content-Length", "0"))
            if not 0 < n <= 4096: raise ValueError()
            data = json.loads(self.rfile.read(n))
            name = data["name"]
            if not isinstance(name, str) or not name: raise ValueError()
            if self.command == "DELETE":
                if name not in STATE: return self.reply(404, {"error": "missing"})
                del STATE[name]
                return self.reply(200, {"deleted": name})
            enabled = data["enabled"]
            description = data.get("description", "")
            if not isinstance(enabled, bool) or not isinstance(description, str): raise ValueError()
            if self.command == "POST" and name in STATE:
                return self.reply(409, {"error": "exists"})
            if self.command == "PUT" and name not in STATE:
                return self.reply(404, {"error": "missing"})
            STATE[name] = {"enabled": enabled, "description": description}
            return self.reply(201 if self.command == "POST" else 200, STATE[name])
        except (ValueError, KeyError, TypeError):
            return self.reply(400, {"error": "invalid JSON or schema"})
    do_GET = request
    do_POST = request
    do_PUT = request
    do_DELETE = request
if __name__ == "__main__":
    HTTPServer(("127.0.0.1", 8765), API).serve_forever()

client.py

import json
from urllib.request import Request, urlopen
from urllib.error import HTTPError
URL = "http://127.0.0.1:8765/api/interfaces"
def call(method, data=None, token="training-token"):
    body = None if data is None else json.dumps(data).encode()
    req = Request(URL, data=body, method=method, headers={"Authorization": "Bearer " + token, "Content-Type": "application/json"})
    try:
        with urlopen(req, timeout=5) as res:
            return res.status, json.load(res)
    except HTTPError as err:
        return err.code, json.load(err)
if __name__ == "__main__":
    print(call("GET", token="wrong"))
    print(call("GET"))
    print(call("POST", {"name": "g0/1", "enabled": False, "description": "TEST"}))
    print(call("PUT", {"name": "g0/1", "enabled": True, "description": "UPLINK"}))
    print(call("GET"))
    print(call("DELETE", {"name": "g0/1"}))
